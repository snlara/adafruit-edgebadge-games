import board
import analogio
import digitalio
import displayio
import terminalio
import random
import time
from adafruit_display_text import label

# 1. Setup Sensors
light_sensor = analogio.AnalogIn(board.A7)
mic = analogio.AnalogIn(board.A2)

# 2. Setup Display & Objects
display = board.DISPLAY
group = displayio.Group()
score = 0
level = 1
num_enemies = 4
text_area = label.Label(terminalio.FONT, text="LVL 1 - SCORE: 0", color=0xFFFFFF, x=5, y=10)
player = label.Label(terminalio.FONT, text="O", color=0x00FF00, x=80, y=100)
group.append(text_area); group.append(player)
display.root_group = group

enemies = []

def spawn_enemies(count):
    """Spawns a specific number of enemies in rows."""
    for i in range(count):
        # Calculate rows and columns
        col = i % 6
        row = i // 6
        e = label.Label(terminalio.FONT, text="X", color=0xFF0000, x=10 + (col * 25), y=20 + (row * 20))
        enemies.append(e)
        group.append(e)

spawn_enemies(num_enemies)
bullet = label.Label(terminalio.FONT, text="|", color=0xFFFF00, x=82, y=120)
group.append(bullet)

# 3. Main Logic
enemy_dx = 2; frame_count = 0
latch = digitalio.DigitalInOut(board.BUTTON_LATCH); latch.switch_to_output(value=True)
clock = digitalio.DigitalInOut(board.BUTTON_CLOCK); clock.switch_to_output(value=False)
data = digitalio.DigitalInOut(board.BUTTON_OUT); data.switch_to_input()

def get_buttons():
    latch.value = False; latch.value = True; buttons = 0
    for i in range(7, -1, -1):
        clock.value = True
        if not data.value: buttons |= (1 << i)
        clock.value = False
    return buttons

bullet_active = False
while True:
    b = get_buttons()
    frame_count += 1
    
    # Movement & Shooting
    if b == 0xfd and player.x > 5: player.x -= 8
    if b == 0xef and player.x < 150: player.x += 8
    if b == 0x7f and not bullet_active:
        bullet.x = player.x + 2; bullet.y = player.y - 5; bullet_active = True
        
    if bullet_active:
        bullet.y -= 10
        if bullet.y < 0: bullet_active = False

    # Move enemies
    if frame_count % 3 == 0:
        move_down = False
        for e in enemies:
            e.x += enemy_dx
            if e.x > 150 or e.x < 5: move_down = True
        if move_down:
            enemy_dx *= -1
            for e in enemies: e.y += 5

    # Check for next level
    if len(enemies) == 0:
        level += 1
        num_enemies += 2
        spawn_enemies(num_enemies)
        text_area.text = f"LVL {level} - SCORE: {score}"

    # Hit Detection
    for e in enemies[:]: # Iterate over a copy to allow removal
        if bullet_active and abs(bullet.x - e.x) < 8 and abs(bullet.y - e.y) < 8:
            bullet_active = False; bullet.y = 120
            score += 10; enemies.remove(e); group.remove(e)
            text_area.text = f"LVL {level} - SCORE: {score}"
            
    # Smart Bomb
    if light_sensor.value < 5000 or mic.value > 45000:
        for e in enemies[:]: enemies.remove(e); group.remove(e)
            
    time.sleep(0.05)
