import board
import digitalio
import displayio
import terminalio
import random
import time
import neopixel  # <--- Added this
from adafruit_display_text import label

# 1. Setup Display & Objects
display = board.DISPLAY
group = displayio.Group()
score = 0

# Initialize NeoPixels (5 lights, low brightness)
pixels = neopixel.NeoPixel(board.NEOPIXEL, 5, brightness=0.05, auto_write=True)
pixels.fill((0, 0, 0)) # Start off

text_area = label.Label(terminalio.FONT, text="SCORE: 0", color=0xFFFFFF, x=5, y=10)
player = label.Label(terminalio.FONT, text="O", color=0x00FF00, x=80, y=100)
enemy = label.Label(terminalio.FONT, text="X", color=0xFF0000, x=random.randint(10, 150), y=0)
bullet = label.Label(terminalio.FONT, text="|", color=0xFFFF00, x=82, y=120)
group.append(text_area); group.append(player); group.append(enemy); group.append(bullet)
display.root_group = group

# 2. Setup Buttons
latch = digitalio.DigitalInOut(board.BUTTON_LATCH); latch.switch_to_output(value=True)
clock = digitalio.DigitalInOut(board.BUTTON_CLOCK); clock.switch_to_output(value=False)
data = digitalio.DigitalInOut(board.BUTTON_OUT); data.switch_to_input()

def get_buttons():
    latch.value = False; latch.value = True; buttons = 0
    for i in range(7, -1, -1):
        clock.value = True
        if not data.value: buttons |= (1 << i)
        clock.value = False
    return buttons

# Flash function
def flash_muzzle():
    pixels.fill((255, 100, 0)) # Orange flash
    time.sleep(0.03)           # Duration of flash
    pixels.fill((0, 0, 0))     # Turn off

# 3. Main Game Loop
bullet_active = False
while True:
    b = get_buttons()
    
    # Movement
    if b == 0xfd and player.x > 5: player.x -= 8
    if b == 0xef and player.x < 150: player.x += 8
    
    # Shoot
    if b == 0x7f and not bullet_active:
        flash_muzzle() # <--- Triggers the light flash when button pressed
        bullet.x = player.x + 2; bullet.y = player.y - 5
        bullet_active = True
        
    # Bullet Logic
    if bullet_active:
        bullet.y -= 10
        if bullet.y < 0: bullet_active = False
        
    # Enemy Logic
    enemy.y += 4
    if enemy.y > 120:
        enemy.y = 0; enemy.x = random.randint(10, 150)
        
    # Collision & Hit Detection
    if bullet_active and abs(bullet.x - enemy.x) < 8 and abs(bullet.y - enemy.y) < 8:
        enemy.y = 0; enemy.x = random.randint(10, 150)
        bullet_active = False; bullet.y = 120
        score += 5; text_area.text = f"SCORE: {score}"; text_area.color = 0x00FF00
        
    # Reset color after hit
    if text_area.color == 0x00FF00:
        time.sleep(0.1); text_area.color = 0xFFFFFF
        
    # Player collision
    if abs(player.x - enemy.x) < 8 and abs(player.y - enemy.y) < 8:
        text_area.text = "GAME OVER!"; text_area.color = 0xFF0000; time.sleep(1)
        score = 0; player.x=80; enemy.y=0; text_area.color = 0xFFFFFF
        
    time.sleep(0.05)
